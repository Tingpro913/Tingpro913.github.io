   345 ```typescript
   346 import ky, { type AfterResponseHook } from "ky";
   347 import { createParser } from "eventsource-parser";
   348 import { useState, useRef } from "react";
   349 
   350 const supabaseUrl = import.meta.env.VITE_SUPABASE_URL;
   351 const supabaseAnonKey = import.meta.env.VITE_SUPABASE_ANON_KEY;
   352 
   353 /**
   354  * 创建 SSE Hook，将流式事件逐块转发给回调。
   355  * @param onChunk - 每收到一个文本增量时调用
   356  * @param onDone - 流结束时调用
   357  * @param onError - 出错时调用
   358  */
   359 function createMiniMaxSSEHook(
   360   onChunk: (text: string) => void,
   361   onDone: () => void,
   362   onError: (err: Error) => void
   363 ): AfterResponseHook {
   364   return async (_request, _opts, response) => {
   365     if (!response.ok || !response.body) return;
   366 
   367     const reader = response.body.getReader();
   368     const decoder = new TextDecoder("utf8");
   369     const parser = createParser({
   370       onEvent: (event) => {
   371         if (!event.data || event.data === "[DONE]") return;
   372         try {
   373           const parsed = JSON.parse(event.data);
   374           const delta = parsed.choices?.[0]?.delta?.content ?? "";
   375           if (delta) onChunk(delta);
   376         } catch { /* 不完整 chunk，跳过 */ }
   377       },
   378     });
   379 
   380     const read = (): void => {
   381       reader.read().then(({ done, value }) => {
   382         if (done) { onDone(); return; }
   383         parser.feed(decoder.decode(value, { stream: true }));
   384         read();
   385       }).catch((err) => onError(err as Error));
   386     };
   387 
   388     read();
   389     return response;
   390   };
   391 }
   392 
   393 /**
   394  * 流式调用 MiniMax 对话补全，实时输出文本增量。
   395  * @param messages - 对话历史
   396  * @param onChunk - 每收到增量文本时调用
   397  * @param onDone - 流结束时调用
   398  * @param signal - AbortSignal，用于取消请求
   399  */
   400 async function streamMiniMaxChat(
   401   messages: Array<{ role: string; content: string; name?: string }>,
   402   onChunk: (text: string) => void,
   403   onDone: () => void,
   404   signal?: AbortSignal
   405 ): Promise<void> {
   406   await ky.post(`${supabaseUrl}/functions/v1/minimax-chat-stream`, {
   407     json: { model: "MiniMax-M2.5", messages },
   408     headers: {
   409       Authorization: `Bearer ${supabaseAnonKey}`,
   410       apikey: supabaseAnonKey,
   411       "Content-Type": "application/json",
   412     },
   413     signal,
   414     hooks: {
   415       afterResponse: [createMiniMaxSSEHook(onChunk, onDone, (err) => { throw err; })],
   416     },
   417   });
   418 }
   419 
   420 // React 组件使用示例
   421 function ChatDemo() {
   422   const [content, setContent] = useState("");
   423   const [loading, setLoading] = useState(false);
   424   const abortRef = useRef<AbortController | null>(null);
   425 
   426   const handleSend = async () => {
   427     setContent("");
   428     setLoading(true);
   429     abortRef.current = new AbortController();
   430 
   431     await streamMiniMaxChat(
   432       [
   433         { role: "system", content: "你是智能助手。", name: "MiniMax AI" },
   434         { role: "user", content: "介绍一下你自己。", name: "用户" },
   435       ],
   436       (chunk) => setContent((prev) => prev + chunk),
   437       () => setLoading(false),
   438       abortRef.current.signal
   439     );
   440   };
   441 
   442   const handleStop = () => abortRef.current?.abort();
   443 
   444   return (
   445     <div>
   446       <button onClick={handleSend} disabled={loading}>发送</button>
   447       <button onClick={handleStop} disabled={!loading}>停止</button>
   448       <p>{content}</p>
   449     </div>
   450   );
   451 }

   489 ## 注意事项
   490 
   491 1. **密钥安全**：`INTEGRATIONS_API_KEY` 仅在 Edge Function 服务端读取，严禁暴露给前端或客户端代码。
   492 2. **错误处理**：
   493    - 务必检查 `base_resp.status_code`（非 0 表示业务错误）；
   494    - 处理 HTTP 429（触发限流）和 402（余额不足）；
   495    - 输入/输出命中敏感词时（`input_sensitive`/`output_sensitive` 为 `true`），回复内容可能为空。
   496 3. **计费**：当前免费，但 `need_count_calls: true` 表示调用次数仍会统计，避免不必要的重复调用。
   497 4. **流式输出**：推理模型（MiniMax-M2.5）建议使用流式输出以获得最佳体验；流式请求结束后会在最后一个 chunk 中附带 `usage` 统计（需设置 `stream_options.include_usage: true`）。
   498 5. **Token 限制**：MiniMax-M2.5 默认 `max_completion_tokens` 为 10240；若生成因 `length` 原因中断，请调高此值。
   499 6. **图文混合输入**：`content` 传数组格式时，`image_url.url` 支持公网 URL 或 Base64 Data URL；MiniProgram 中建议使用公网 URL。
   500 7. **工具调用**：使用 `tools` + `tool_choice: "auto"` 时，模型返回 `finish_reason: "tool_calls"`，需解析 `choices[].message.tool_calls` 并将工具执行结果以 `role: "tool"` 消息追加后继续调用。
